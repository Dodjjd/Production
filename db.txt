--изделия
CREATE TABLE products (
    prod_id INTEGER PRIMARY KEY,
    prod_name VARCHAR(100) NOT NULL,
    selling_price NUMERIC(10,2) DEFAULT 0 CHECK (selling_price >= 0)
);

--комплектующие
CREATE TABLE components (
    comp_id INTEGER PRIMARY KEY,
    comp_name VARCHAR(100) NOT NULL,
    unit_price NUMERIC(10,2) DEFAULT 0 CHECK (unit_price >= 0),
    tech_process_id INTEGER
);

--Справочник тех операций
CREATE TABLE tech_ops (
    op_id INTEGER PRIMARY KEY,
    op_name VARCHAR(100) NOT NULL,
    hours_per_unit NUMERIC(8,2) CHECK (hours_per_unit >= 0),
    cost_per_hour NUMERIC(10,2) DEFAULT 500.00 CHECK (cost_per_hour >= 0)
);

--материалы
CREATE TABLE materials (
    mat_id INTEGER PRIMARY KEY,
    mat_name VARCHAR(100) UNIQUE NOT NULL,
    unit_price NUMERIC(10,2) DEFAULT 0 CHECK (unit_price >= 0)
);

--техпроцесс
CREATE TABLE tech_process (
    tp_id SERIAL PRIMARY KEY,
    comp_id INTEGER REFERENCES components(comp_id) ON DELETE CASCADE,
    mat_id INTEGER REFERENCES materials(mat_id),
    mat_qty NUMERIC(8,2) CHECK (mat_qty > 0),
    op_id INTEGER REFERENCES tech_ops(op_id),
    op_sequence INTEGER
);

--состав изделия
CREATE TABLE product_components (
    prod_id INTEGER REFERENCES products(prod_id) ON DELETE CASCADE,
    comp_id INTEGER REFERENCES components(comp_id),
    qty_required NUMERIC(8,2) CHECK (qty_required > 0),
    PRIMARY KEY (prod_id, comp_id)
);

--продажи
CREATE TABLE sales (
    sale_id SERIAL PRIMARY KEY,
    prod_id INTEGER REFERENCES products(prod_id),
    sale_date DATE DEFAULT CURRENT_DATE,
    quantity INTEGER DEFAULT 1 CHECK (quantity > 0),
    discount NUMERIC(5,2) DEFAULT 0 CHECK (discount >= 0 AND discount <= 100)
);

CREATE INDEX idx_comp_tech ON components(tech_process_id);
CREATE INDEX idx_tp_comp ON tech_process(comp_id);
CREATE INDEX idx_tp_mat ON tech_process(mat_id);
CREATE INDEX idx_sales_prod ON sales(prod_id);
CREATE INDEX idx_sales_date ON sales(sale_date);

--проекция по одной таблице
CREATE VIEW expensive_materials AS
SELECT mat_name, unit_price
FROM materials
WHERE unit_price > 1000;


-- проекция по нескольким таблицам(состав изделия с ценами)
CREATE VIEW product_full_cost AS
SELECT p.prod_id, p.prod_name,
       pc.comp_id, c.comp_name, pc.qty_required,
       c.unit_price AS comp_price,
       (pc.qty_required * c.unit_price) AS comp_cost
FROM products p
JOIN product_components pc ON p.prod_id = pc.prod_id
JOIN components c ON pc.comp_id = c.comp_id;

--с GROUP BY и HAVING
CREATE VIEW high_material_components AS
SELECT c.comp_name, SUM(tp.mat_qty) AS total_material_qty
FROM components c
JOIN tech_process tp ON c.comp_id = tp.comp_id
GROUP BY c.comp_name
HAVING SUM(tp.mat_qty) > 2;

--после вставки комплектующего автоматически добавляем типовой техпроцесс
CREATE OR REPLACE FUNCTION auto_create_tech_process()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO tech_process (comp_id, mat_id, mat_qty, op_id, op_sequence)
    VALUES (NEW.comp_id, 1, 1.0, 1, 1);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_auto_tech_process
AFTER INSERT ON components
FOR EACH ROW
EXECUTE FUNCTION auto_create_tech_process();

--нельзя удалить комплектующее, если оно используется в изделии
CREATE OR REPLACE FUNCTION prevent_component_delete()
RETURNS TRIGGER AS $$
BEGIN
    IF EXISTS (SELECT 1 FROM product_components WHERE comp_id = OLD.comp_id) THEN
        RAISE EXCEPTION 'Нельзя удалить комплектующее, которое используется в изделиях';
    END IF;
    RETURN OLD;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_prevent_comp_delete
BEFORE DELETE ON components
FOR EACH ROW
EXECUTE FUNCTION prevent_component_delete();
